+//
+// Two shapes, both real on this rig's own history:
+//   (a) blockedBy names a CANCELLED id - the prerequisite was dropped, so the gate is a permanent
+//       park. `workable()` (src/rig.mjs) refuses the row POSITIVELY (`gate-cancelled:#<id>`), so the
+//       driver never wakes the seat and nobody ever sees the strand (#3237 read as "blocked" while
+//       being permanently unworkable - the third instance in a day).
+//   (b) blockedBy names an id that resolves to NO row. Mint-time validation (#7001) prevents this
+//       going forward, but it cannot fix rows written before the reference check existed. Here the
+//       claim predicate's CONSERVATIVE default passes the unknown id, so the row looks claimable
+//       while its recorded gate is fiction.
+//
+// Detection uses the ONE resolver (gateStates / unsatisfiableGates, src/queue-lifecycle.mjs), never a

+import { workable } from "../src/rig.mjs";

+let failures = 0;

+		failures++;
