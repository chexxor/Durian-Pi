+/**
+ * `release <item-id>` / `hold <item-id> [reason]` (#6102) — the operator's queue-state verbs.
+ *
+ * Why they exist: a parked row could only be un-parked by a SEAT, so the operator clearing a backlog
+ * had no verb for it and no in-process way to show the resident broker saw the change. With a resident
+ * broker these ride the SAME spool path as pause/unpause — producer writes a request, the broker applies
+ * it with its own harness, and the commit that lands is the one the scheduler is subscribed to, which is
+ * what makes "release wakes the seat" observable in the broker log rather than inferred. With no
+ * resident broker this process is the only writer and applies the identical transition directly
+ * (src/broker.mjs applyQueueStateOp), so the two paths cannot drift.
+ */
+async function cmdQueueState(op, rest) {
+  const itemId = Number(rest[0]);
+  if (!Number.isInteger(itemId) || itemId <= 0) fail(`${op} needs an item id: node cli.mjs ${op} <item-id>`);
+  const reason = rest.slice(1).filter((a) => !a.startsWith("--")).join(" ") || undefined;
+  const broker = readBrokerState(DB_PATH);
+  if (brokerAlive(broker)) {
+    const requestId = randomUUID();
+    writeIntent(DB_PATH, {
+      op,
+      itemId,
+      from: "operator",
+      ...(reason ? { reason } : {}),
+      requestId,
+      createdAt: new Date().toISOString(),
+    });
+    const outcome = await awaitIntentOutcome(DB_PATH, requestId);
+    if (outcome.kind === "refused") fail(`${op} #${itemId}: ${outcome.reason}`);
+    if (outcome.kind === "timeout")
+      fail(`${op} #${itemId}: timed out waiting for the resident broker (pid ${broker.pid}) to apply the request`);
+    out({ ok: true, command: op, db: DB_PATH, appliedBy: "broker", requestId, itemId, result: outcome.result });
+    return;
+  }
+  // No resident broker: this process is the only writer, so it may apply the change itself.
+  const { harness } = await openHarness({ enforceSeatRoot: true });
+  try {
+    const result = await applyQueueStateOp(harness, { op, itemId, reason, by: "operator" }, BACKGROUND_CONTEXT);
+    if (!result.ok) fail(`${op} #${itemId}: ${result.detail}`);
+    out({ ok: true, command: op, db: DB_PATH, appliedBy: "this-process", itemId, result });
+  } finally {

+  }

+/**
+ * Poll for EITHER outcome of an intent: the applied result, or the dead-letter reason. A refusal is an
+ * outcome the operator must see, so waiting only on the result file would hang on a gate refusal.
+ */
+async function awaitIntentOutcome(dbPath, requestId, { timeoutMs = 30_000, pollMs = 100 } = {}) {
+  const deadline = Date.now() + timeoutMs;
+  for (;;) {
+    const result = readIntentResult(dbPath, requestId);
+    if (result) return { kind: "applied", result };
+    const error = readIntentError(dbPath, requestId);
+    if (error) return { kind: "refused", reason: error.reason || "dead-lettered", refusal: error.refusal ?? null };
+    if (Date.now() >= deadline) return { kind: "timeout" };
+    await new Promise((r) => setTimeout(r, pollMs));
+  }

+export function parkItem(item, by, reason) {
+	if (isTerminalItem(item))
