+// WHY THIS FILE EXISTS. `send_work` is the ONLY place `blockedBy` can be written, and it writes it
+// once, at creation. `revise_work` touches `subject` and `body` only. So a gate that is wrong,

+// releasing them — it is writing the gate INTO them. Without a write path, every gate stays prose,
+// and prose is the unqueryable form the field was introduced to replace.

+// THREE OUTCOMES the write path must distinguish, because collapsing them is the defect:
+//   SET     — a gate that was not there is now there;

+/**
+ * Validate a gate BEFORE it is written to an EXISTING item (#3570).
+ *
+ * `send_work` can only gate an item at creation, and ids are assigned in increasing order, so a
+ * creation-time gate is trivially safe: it can only name an item that already exists and cannot
+ * name itself. A gate written AFTER the fact has none of those guarantees, so the write path has
+ * to check four things or it stores a row that can never become workable, silently:
+ *   prose     — "GATED on #1" is not a gate (R5 rejects it at the field boundary);
+ *   dangling  — an id that names no queue item is a typo that reads as "unblocked" forever,
+ *               because `workable()`'s conservative default passes unknown ids over;
+ *   self      — an item gated on itself never closes: it is a permanent starvation of its own slot;
+ *   cycle     — A gated on B, B gated on A: neither can ever be workable, and the queue LOOKS
+ *               correctly gated while nothing can run.
+ * Returns `{ ok, gate, error }`; `gate` is the normalized list (null = ungated) so the caller
+ * stores exactly what was validated. `error` is the legible refusal, never a stack.
+ */
+export function validateGate(item, gate, items = []) {
+	let normalized;
+	try {
+		normalized = normalizeBlockedBy(gate);
+	} catch (err) {
+		return { ok: false, gate: null, error: err.message };
+	}
+	if (normalized === null) return { ok: true, gate: null, error: null };
+	const byId = new Map((items ?? []).map((it) => [it.id, it]));
+	for (const id of normalized) {
+		if (id === item.id) {
+			return { ok: false, gate: normalized, error: `item #${item.id} cannot gate itself — a self-gate never closes` };
+		}
+		if (!byId.has(id)) {
+			return {
+				ok: false,
+				gate: normalized,
+				error: `gate #${id} names no queue item — every blockedBy id must point at an item that exists`,
+			};
+		}
+	}
+	// Walk the gate graph FROM the proposed gate. If we can get back to the item being edited, the
+	// gate closes a cycle. This is a plain DFS over a graph whose only edges are `blockedBy` lists.
+	const stack = [...normalized];
+	const seen = new Set();
+	while (stack.length) {
+		const id = stack.pop();
+		if (id === item.id) {
+			return {
+				ok: false,
+				gate: normalized,
+				error: `the gate on item #${item.id} closes a CYCLE through #${[...seen].join(", ")} — every row in a cycle waits on another row in it, so none of them can ever run`,
+			};
+		}
+		if (seen.has(id)) continue;
+		seen.add(id);
+		const target = byId.get(id);
+		for (const next of target?.blockedBy ?? []) if (!seen.has(next)) stack.push(next);
+	}
+	return { ok: true, gate: normalized, error: null };
+}
